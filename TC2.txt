#1. Display all customers along with their account details
SELECT c.*, a.*
FROM Customers c
LEFT JOIN Accounts a
ON c.CustomerID = a.CustomerID;

#2. List customers and the number of accounts they hold

SELECT c.CustomerID, c.CustomerName, COUNT(a.AccountID) AS Account_Count
FROM Customers c
LEFT JOIN Accounts a
ON c.CustomerID = a.CustomerID
GROUP BY c.CustomerID, c.CustomerName;

#3. Show customers who do not have any account


SELECT c.*
FROM Customers c
LEFT JOIN Accounts a
ON c.CustomerID = a.CustomerID
WHERE a.AccountID IS NULL;

#4. Display all accounts including missing customer info


SELECT a.*, c.CustomerName
FROM Accounts a
LEFT JOIN Customers c
ON a.CustomerID = c.CustomerID;

#5. List customers with total balance across accounts

SELECT c.CustomerName, SUM(a.Balance) AS Total_Balance
FROM Customers c
JOIN Accounts a
ON c.CustomerID = a.CustomerID
GROUP BY c.CustomerName;

#6. Show customers and their accounts where account type is “Savings”


SELECT c.CustomerName, a.AccountID, a.Balance
FROM Customers c
JOIN Accounts a
ON c.CustomerID = a.CustomerID
WHERE a.AccountType = 'Savings';

#7. Find customers who opened an account after 2021-01-01

SELECT DISTINCT c.CustomerName
FROM Customers c
JOIN Accounts a
ON c.CustomerID = a.CustomerID
WHERE a.OpenDate > '2021-01-01';

#8. Display accounts in Mumbai Branch along with customer names


SELECT c.CustomerName, a.AccountID, a.Balance
FROM Customers c
JOIN Accounts a
ON c.CustomerID = a.CustomerID
WHERE a.Branch = 'Mumbai Branch';

#9. Show all transactions along with account details


SELECT t.*, a.AccountType, a.Branch
FROM Transactions t
JOIN Accounts a
ON t.AccountID = a.AccountID;

10. Display withdrawal transactions only

(Transactions only)

SELECT *
FROM Transactions
WHERE TransactionType = 'Withdrawal';

#11. Find accounts that have no transactions


SELECT a.*
FROM Accounts a
LEFT JOIN Transactions t
ON a.AccountID = t.AccountID
WHERE t.TransactionID IS NULL;

#12. Display total deposits per account

SELECT AccountID, SUM(Amount) AS Total_Deposits
FROM Transactions
WHERE TransactionType = 'Deposit'
GROUP BY AccountID;

#13. Show maximum transaction amount per account


SELECT AccountID, MAX(Amount) AS Max_Amount
FROM Transactions
GROUP BY AccountID;

#14. Find accounts with balance > 100,000 and list transactions

SELECT a.AccountID, a.Balance, t.TransactionType, t.Amount
FROM Accounts a
JOIN Transactions t
ON a.AccountID = t.AccountID
WHERE a.Balance > 100000;

#15. List transactions in Pune Branch along with account type

SELECT t.TransactionID, t.TransactionType, t.Amount, a.AccountType
FROM Transactions t
JOIN Accounts a
ON t.AccountID = a.AccountID
WHERE a.Branch = 'Pune Branch';

#16. Find accounts where total withdrawals exceed 50,000


SELECT AccountID, SUM(Amount) AS Total_Withdrawals
FROM Transactions
WHERE TransactionType = 'Withdrawal'
GROUP BY AccountID
HAVING SUM(Amount) > 50000;

#17. Show all transactions for each customer


SELECT c.CustomerName, a.AccountID
FROM Customers c
JOIN Accounts a
ON c.CustomerID = a.CustomerID
WHERE a.AccountID IN (SELECT AccountID FROM Transactions);

#18. Display customers who made deposits only


SELECT DISTINCT c.CustomerName
FROM Customers c
JOIN Accounts a
ON c.CustomerID = a.CustomerID
WHERE a.AccountID IN (
    SELECT AccountID
    FROM Transactions
    GROUP BY AccountID
    HAVING SUM(CASE WHEN TransactionType = 'Withdrawal' THEN 1 ELSE 0 END) = 0
);

#19. Find customers older than 35 years who made any transaction


SELECT DISTINCT c.CustomerName, c.Age
FROM Customers c
JOIN Accounts a
ON c.CustomerID = a.CustomerID
WHERE c.Age > 35
AND a.AccountID IN (SELECT AccountID FROM Transactions);

#20. List customers whose total transaction amount exceeds 100,000


SELECT c.CustomerName
FROM Customers c
JOIN Accounts a
ON c.CustomerID = a.CustomerID
WHERE a.AccountID IN (
    SELECT AccountID
    FROM Transactions
    GROUP BY AccountID
    HAVING SUM(Amount) > 100000
);

#21. Display all customers with their last transaction date

SELECT c.CustomerName,
       (SELECT MAX(TransactionDate)
        FROM Transactions t
        WHERE t.AccountID = a.AccountID) AS Last_Transaction_Date
FROM Customers c
LEFT JOIN Accounts a
ON c.CustomerID = a.CustomerID;
